--- layout: default title: "Reference Vector" ---
Create 
-- vector2(...) is a GLOBAL constructor registered by ARGoS

v = vector2()          -- (0,0) : default constructor
v = vector2(1.5, -3)   -- (x,y) : value constructor
u = vector2(v)         -- copy constructor (useful before in-place ops)
Access (x,y) 
v = vector2(3, 4)

-- read
log("v.x = " .. v.x)
log("v.y = " .. v.y)

-- write (only 'x' and 'y' are valid keys)
v.x = 10
v.y = -2
Print & compare 
v = vector2(1, 2)
w = vector2(1, 2)
z = vector2(2, 1)

-- tostring() is implemented, so this works:
log("v = " .. tostring(v))
log("v = " .. v)  -- equivalent

-- equality
log(v == w)  -- true
log(v == z)  -- false
Operators 
a = vector2(1, 2)
b = vector2(3, 4)

s = a + b     -- addition
d = a - b     -- subtraction
n = -a        -- unary minus (negation)

-- scalar multiplication (both orders are supported)
k1 = a * 10
k2 = 10 * a

-- NOTE: vector * vector is NOT supported (will error)
Length 
v = vector2(3, 4)
log(v:length())   -- 5
Normalize (in-place) 
-- normalize() MODIFIES the vector object and returns it

v = vector2(3, 4)
v:normalize()
log(v:length())   -- ~1

-- if you want a normalized COPY:
v = vector2(3, 4)
u = vector2(v):normalize()  -- copy then normalize
Dot product 
a = vector2(1, 0)
b = vector2(0, 1)

d = a:dot(b)   -- returns a NUMBER
log(d)         -- 0
Cross product (in-place) 
-- cross() MODIFIES the first vector and returns it
-- (exact behavior depends on ARGoS CVector2::CrossProduct)

a = vector2(1, 0)
b = vector2(0, 1)

a:cross(b)     -- 'a' is now modified

-- if you want to keep 'a' unchanged:
a = vector2(1, 0)
c = vector2(a):cross(b)   -- cross on a copy
Rotate (in-place, radians) 
-- rotate(angle) expects angle in RADIANS
-- rotate() MODIFIES the vector and returns it

v = vector2(1, 0)
v:rotate(math.pi / 2)   -- ~ (0,1)

-- rotated COPY:
v = vector2(1, 0)
w = vector2(v):rotate(math.pi / 2)
Common patterns (ARGoS control) 
-- Bearing (radians) -> unit direction vector
function bearing_to_vec(bearing)
  return vector2(1, 0):rotate(bearing)
end

-- Safe normalize (avoid normalizing zero vector)
function normalize_safe(v)
  if v:length() > 0 then
    v:normalize()
  end
  return v
end

-- Sum directions of received RAB packets (example)
function step()
  local sum = vector2()
  for _, p in ipairs(robot.range_and_bearing) do
    local dir = bearing_to_vec(p.bearing)
    sum = sum + dir
  end

  normalize_safe(sum)
  -- sum.x / sum.y is now a heading (or (0,0) if no neighbors)
end
